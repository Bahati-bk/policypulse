    47→    47→function getFileIcon(fileName: string) {
    48→    48→  const ext = fileName?.split('.').pop()?.toLowerCase()
    49→    49→  if (ext === 'pdf') return <File className="h-4 w-4 text-rose-500" />
    50→    50→  if (ext === 'docx' || ext === 'doc') return <FileSpreadsheet className="h-4 w-4 text-teal-500" />
    51→    51→  if (ext === 'txt') return <FileText className="h-4 w-4 text-slate-500" />
    52→    52→  return <FileText className="h-4 w-4 text-muted-foreground" />
    53→    53→}

    55→    55→function formatFileSize(bytes: number): string {
    56→    56→  if (bytes < 1024) return bytes + ' B'
    57→    57→  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB'
    58→    58→  return (bytes / (1024 * 1024)).toFixed(1) + ' MB'
    59→    59→}
