   176→   176→        method: 'POST',
   177→   177→        headers: { 'Content-Type': 'application/json' },

   179→   179→      })

   181→   181→        const d = await res.json()

   184→   184→      return res.json()

   196→   196→    mutationFn: async (id: string) => {

   199→   199→      return res.json()
