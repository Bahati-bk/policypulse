   139→// ── Component ────────────────────────────────────────────────────────
   140→
   141→export default function AlertsView() {
   142→  const user = useAppStore(s => s.user)
   143→  const queryClient = useQueryClient()
   144→  const [statusFilter, setStatusFilter] = useState('')
   145→  const [search, setSearch] = useState('')
   146→  const [selectedAlert, setSelectedAlert] = useState<Alert | null>(null)
   147→  const [rejectDialog, setRejectDialog] = useState<string | null>(null)
   148→  const [rejectReason, setRejectReason] = useState('')
   149→  const [createOpen, setCreateOpen] = useState(false)
   150→  const [ussdDialog, setUssdDialog] = useState<string | null>(null)
   151→  const [ussdPhones, setUssdPhones] = useState('')
   152→  const [form, setForm] = useState<FormData>(emptyForm)
   153→
   154→  // ── Queries ──────────────────────────────────────────────────────
   155→
   156→  const { data: alertsData, isLoading } = useQuery<Alert[]>({
   157→    queryKey: ['alerts', statusFilter],
   158→    queryFn: () => fetch(`/api/alerts${statusFilter ? `?status=${statusFilter}` : ''}`).then(r => r.json()),
   159→    enabled: !!user,
   160→  })
   161→  const alerts = safeArray<Alert>(alertsData)
   162→
   163→  // ── Counts for filter badges ─────────────────────────────────────
   164→
   165→  const statusCounts = useMemo(() => {
   166→    const counts: Record<string, number> = { '': alerts.length }
   167→    for (const a of alerts) {
   168→      counts[a.status] = (counts[a.status] || 0) + 1
   169→    }
   170→    return counts
   171→  }, [alerts])
   172→
   173→  // ── Mutations ────────────────────────────────────────────────────
   174→
   175→  const createMutation = useMutation({
   176→    mutationFn: async (data: FormData) => {
   177→      const res = await fetch('/api/alerts', {
   178→        method: 'POST',
   179→        headers: { 'Content-Type': 'application/json' },
   180→        body: JSON.stringify(data),
   181→      })
   182→      if (!res.ok) {
   183→        const d = await res.json()
   184→        throw new Error(d.error || 'Failed to create alert')
   185→      }
   186→      return res.json()
   187→    },
   188→    onSuccess: () => {
   189→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   190→      toast.success('Alert created successfully')
   191→      setCreateOpen(false)
   192→      setForm(emptyForm)
   193→    },
   194→    onError: (e) => toast.error(e.message),
   195→  })
   196→
   197→  const approveMutation = useMutation({
   198→    mutationFn: async (id: string) => {
   199→      const res = await fetch(`/api/alerts/${id}/approve`, { method: 'POST' })
   200→      if (!res.ok) throw new Error('Failed')
   201→      return res.json()
   202→    },
   203→    onSuccess: () => {
   204→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   205→      toast.success('Alert approved')
   206→      setSelectedAlert(null)
   207→    },
   208→    onError: () => toast.error('Failed to approve'),
   209→  })
   210→
   211→  const rejectMutation = useMutation({
   212→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   213→      const res = await fetch(`/api/alerts/${id}/reject`, {
   214→        method: 'POST',
   215→        headers: { 'Content-Type': 'application/json' },
   216→        body: JSON.stringify({ reason }),
   217→      })
   218→      if (!res.ok) throw new Error('Failed')
   219→      return res.json()
   220→    },
   221→    onSuccess: () => {
   222→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   223→      toast.success('Alert rejected')
   224→      setRejectDialog(null)
   225→      setSelectedAlert(null)
   226→    },
   227→    onError: () => toast.error('Failed to reject'),
   228→  })
   229→
   230→  const sendMutation = useMutation({
   231→    mutationFn: async (id: string) => {
   232→      const res = await fetch(`/api/alerts/${id}/send`, { method: 'POST' })
   233→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Failed') }
   234→      return res.json()
   235→    },
   236→    onSuccess: (data) => {
   237→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   238→      queryClient.invalidateQueries({ queryKey: ['notifications'] })
   239→      toast.success(`Notifications sent to ${data.sentTo} users`)
   240→      setSelectedAlert(null)
   241→    },
   242→    onError: (e) => toast.error(e.message),
   243→  })
   244→
   245→  const ussdMutation = useMutation({
   246→    mutationFn: async ({ alertId, phoneNumbers, message }: { alertId: string; phoneNumbers: string[]; message: string }) => {
   247→      const res = await fetch('/api/ussd/send-alert', {
   248→        method: 'POST',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify({ phoneNumbers, message, alertId }),
   251→      })
   252→      if (!res.ok) {
   253→        const d = await res.json()
   254→        throw new Error(d.error || 'Failed to send USSD')
   255→      }
   256→      return res.json()
   257→    },
   258→    onSuccess: (data) => {
   259→      toast.success(`USSD alert queued to ${data.messageCount} recipient(s) via *384*17818#`)
   260→      setUssdDialog(null)
   261→      setUssdPhones('')
   262→    },
   263→    onError: (e) => toast.error(e.message),
   264→  })
   265→
   266→  // ── Helpers ──────────────────────────────────────────────────────
   267→
   268→  const isAdmin = user?.role === 'ADMIN'
   269→
   270→  function handleUssdSend() {
   271→    if (!selectedAlert) return
   272→    const phones = ussdPhones
   273→      .split(/[,\n;\s]+/)
   274→      .map(p => p.trim())
   275→      .filter(p => p.length > 0)
   276→    if (phones.length === 0) {
   277→      toast.error('Enter at least one phone number')
   278→      return
   279→    }
   280→    const message = `[PolicyPulse] ${selectedAlert.title}
   281→${selectedAlert.summary || ''}`.trim()
   282→    ussdMutation.mutate({ alertId: selectedAlert.id, phoneNumbers: phones, message })
   283→  }
   284→
   285→  const filtered = alerts.filter((alert) => {
   286→    if (!search) return true
   287→    const q = search.toLowerCase()
   288→    return (
   289→      alert.title?.toLowerCase().includes(q) ||
   290→      alert.summary?.toLowerCase().includes(q) ||
   291→      alert.whatChanged?.toLowerCase().includes(q)
   292→    )
   293→  })
   294→
   295→  function updateField<K extends keyof FormData>(key: K, value: FormData[K]) {
   296→    setForm(prev => ({ ...prev, [key]: value }))
   297→  }
   298→
   299→  function resetForm() {
   300→    setForm(emptyForm)
   301→  }
   302→
   303→  function handleSubmit(e: React.FormEvent) {
   304→    e.preventDefault()
   305→    if (!form.title.trim() || form.summary.trim().length < 10) return
   306→    createMutation.mutate(form)
   307→  }
   308→
   309→  function exportCSV() {
   310→    const headers = ['Title', 'Status', 'Severity', 'Change Type', 'Created', 'Summary', 'What Changed', 'Who Is Affected']
   311→    const rows = filtered.map(a => [
   312→      `"${(a.title || '').replace(/"/g, '\"')}"`,
   313→      a.status,
   314→      a.severity || '',
   315→      a.changeType || '',
   316→      a.createdAt ? format(new Date(a.createdAt), 'yyyy-MM-dd') : '',
   317→      `"${(a.summary || '').replace(/"/g, '\"')}"`,
   318→      `"${(a.whatChanged || '').replace(/"/g, '\"')}"`,
   319→      `"${(a.whoIsAffected || '').replace(/"/g, '\"')}"`,
   320→    ])
   321→    const csv = [headers.join(','), ...rows.map(r => r.join(','))].join('\n')
   322→    const blob = new Blob([csv], { type: 'text/csv' })
   323→    const url = URL.createObjectURL(blob)
   324→    const a = document.createElement('a')
   325→    a.href = url
   326→    a.download = `policypulse-alerts-${format(new Date(), 'yyyy-MM-dd')}.csv`
   327→    a.click()
   328→    URL.revokeObjectURL(url)
   329→    toast.success('Alerts exported to CSV')
   330→  }
   331→
   332→  // ── Render ───────────────────────────────────────────────────────
   333→
   334→  return (
   335→    <motion.div variants={container} initial="hidden" animate="show" className="space-y-6">
   336→      {/* ── Header ── */}
   337→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   338→        <div>
   339→          <h1 className="text-2xl font-bold tracking-tight">Alerts</h1>
   340→          <p className="text-muted-foreground text-sm">Policy change alerts and notifications</p>
   341→        </div>
   342→        <div className="flex items-center gap-2">
   343→          {filtered.length > 0 && (
   344→            <Button variant="outline" size="sm" onClick={exportCSV}>
   345→              <Download className="mr-2 h-3.5 w-3.5" />Export CSV
   346→            </Button>
   347→          )}
   348→          <Button size="sm" onClick={() => setCreateOpen(true)}>
   349→            <Plus className="mr-2 h-3.5 w-3.5" />Create Alert
   350→          </Button>
   351→        </div>
   352→      </div>
   353→
   354→      {/* ── Filters ── */}
   355→      <div className="flex flex-col sm:flex-row gap-3">
   356→        <div className="relative flex-1">
   357→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   358→          <Input
   359→            placeholder="Search alerts by title, summary, or changes..."
   360→            value={search}
   361→            onChange={e => setSearch(e.target.value)}
   362→            className="pl-9"
   363→          />
   364→        </div>
   365→        <div className="flex gap-2 flex-wrap">
   366→          {statusFilterOptions.map(s => {
   367→            const count = statusCounts[s] || 0
   368→            return (
   369→              <Button
   370→                key={s}
   371→                variant={statusFilter === s ? 'default' : 'outline'}
   372→                size="sm"
   373→                onClick={() => setStatusFilter(s)}
   374→                className="gap-1.5"
   375→              >
   376→                {s ? s.replace(/_/g, ' ') : 'All'}
   377→                {count > 0 && (
   378→                  <span className={
   379→                    statusFilter === s
   380→                      ? 'bg-background/20 text-current text-xs px-1.5 py-0.5 rounded-full min-w-[20px] text-center'
   381→                      : 'bg-muted text-muted-foreground text-xs px-1.5 py-0.5 rounded-full min-w-[20px] text-center'
   382→                  }>
   383→                    {count}
   384→                  </span>
   385→                )}
   386→              </Button>
   387→            )
   388→          })}
   389→        </div>
   390→      </div>
   391→
   392→      {/* ── Alert List ── */}
   393→      {isLoading ? (
   394→        <div className="space-y-4">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-40 rounded-lg" />)}</div>
   395→      ) : filtered.length === 0 ? (
   396→        <Card>
   397→          <CardContent className="py-16 px-8 text-center">
   398→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-muted">
   399→              <Bell className="h-8 w-8 text-muted-foreground/50" />
   400→            </div>
   401→            <p className="font-semibold text-muted-foreground text-lg">No alerts found</p>
   402→            <p className="text-sm text-muted-foreground/70 mt-2 max-w-md mx-auto">
   403→              {search || statusFilter
   404→                ? 'Try adjusting your search or filter to find what you\'re looking for.'
   405→                : 'Alerts will appear here when policy changes are detected or manually created. Get started by creating your first alert.'}
   406→            </p>
   407→            {!search && !statusFilter && (
   408→              <Button className="mt-4" size="sm" onClick={() => setCreateOpen(true)}>
   409→                <Plus className="mr-2 h-3.5 w-3.5" />Create Alert
   410→              </Button>
   411→            )}
   412→          </CardContent>
   413→        </Card>
   414→      ) : (
   415→        <motion.div variants={container} className="space-y-3">
   416→          {filtered.map((alert) => (
   417→            <motion.div key={alert.id} variants={item}>
   418→              <Card
   419→                className={`hover:shadow-md transition-all cursor-pointer group border-l-4 ${severityBorder[alert.severity] || 'border-l-slate-300'}`}
   420→                onClick={() => setSelectedAlert(alert)}
   421→              >
   422→                <CardContent className="p-4">
   423→                  <div className="flex items-start justify-between gap-3">
   424→                    <div className="flex items-start gap-3 min-w-0">
   425→                      {/* Severity dot */}
   426→                      <div className="relative shrink-0 mt-1">
   427→                        <div className={`h-2.5 w-2.5 rounded-full ${severityColors[alert.severity] || 'bg-slate-400'}`} />
   428→                        {alert.severity === 'CRITICAL' && (
   429→                          <span className="absolute inset-0 rounded-full bg-rose-400 animate-ping opacity-75" />
   430→                        )}
   431→                      </div>
   432→                      <div className="min-w-0">
   433→                        <p className="font-medium text-sm">{alert.title}</p>
   434→                        {alert.summary && (
   435→                          <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{alert.summary}</p>
   436→                        )}
   437→                        <div className="flex items-center gap-2 mt-2 flex-wrap">
   438→                          <Badge variant="outline" className={statusColors[alert.status] || ''}>
   439→                            {alert.status.replace(/_/g, ' ')}
   440→                          </Badge>
   441→                          <Badge variant="outline" className={severityBadge[alert.severity] || ''}>
   442→                            {alert.severity}
   443→                          </Badge>
   444→                          {alert.changeType && (
   445→                            <Badge variant="outline" className={changeTypeBadge[alert.changeType] || ''}>
   446→                              {changeTypeLabel[alert.changeType] || alert.changeType.replace(/_/g, ' ')}
   447→                            </Badge>
   448→                          )}
   449→                        </div>
   450→                      </div>
   451→                    </div>
   452→                    <span className="text-xs text-muted-foreground shrink-0">
   453→                      {alert.createdAt ? format(new Date(alert.createdAt), 'MMM d, yyyy') : ''}
   454→                    </span>
   455→                  </div>
   456→                </CardContent>
   457→              </Card>
   458→            </motion.div>
   459→          ))}
   460→        </motion.div>
   461→      )}
   462→
   463→      {/* ── Create Alert Dialog ── */}
   464→      <Dialog open={createOpen} onOpenChange={(open) => { if (!open) { setCreateOpen(false); resetForm() } }}>
   465→        <DialogContent className="max-w-2xl max-h-[90vh] pointer-events-auto">
   466→          <DialogHeader>
   467→            <DialogTitle className="flex items-center gap-2">
   468→              <Zap className="h-5 w-5 text-amber-500" />
   469→              Create New Alert
   470→            </DialogTitle>
   471→            <DialogDescription>
   472→              Fill in the details to create a new policy change alert. The alert will be created as a draft.
   473→            </DialogDescription>
   474→          </DialogHeader>
   475→
   476→          <form onSubmit={handleSubmit}>
   477→            <ScrollArea className="max-h-[65vh] pr-4">
   478→              <div className="space-y-5 pb-4">
   479→                {/* Title */}
   480→                <div className="space-y-2">
   481→                  <Label htmlFor="alert-title">Title <span className="text-red-500">*</span></Label>
   482→                  <Input
   483→                    id="alert-title"
   484→                    placeholder="e.g. NEW: Digital Service Tax Amendment"
   485→                    value={form.title}
   486→                    onChange={e => updateField('title', e.target.value)}
   487→                    required
   488→                  />
   489→                </div>
   490→
   491→                {/* Summary */}
   492→                <div className="space-y-2">
   493→                  <div className="flex items-center justify-between">
   494→                    <Label htmlFor="alert-summary">Summary <span className="text-red-500">*</span></Label>
   495→                    <span className={`text-xs ${form.summary.length >= 10 ? 'text-muted-foreground' : 'text-amber-500'}`}>
   496→                      {form.summary.length}/10 min characters
   497→                    </span>
   498→                  </div>
   499→                  <Textarea
   500→                    id="alert-summary"
   501→                    placeholder="Brief summary of the policy change..."
   502→                    value={form.summary}
   503→                    onChange={e => updateField('summary', e.target.value)}
   504→                    rows={3}
   505→                    required
   506→                  />
   507→                </div>
