    44→    44→function readFileAsBase64(file: File): Promise<string> {
    45→    45→  return new Promise((resolve, reject) => {
    46→    46→    const reader = new FileReader()
    47→    47→    reader.onload = () => {
    48→    48→      const result = reader.result as string
    49→    49→      resolve(result.split(',')[1])
    50→    50→    }
    51→    51→    reader.onerror = reject
    52→    52→    reader.readAsDataURL(file)
    53→    53→  })
    54→    54→}
