    81→function getInitials(name: string | null | undefined): string {
    82→  if (!name) return 'U'
    83→  const clean = name.replace(/ \(Deactivated\)$/, '')
    84→  const parts = clean.split(' ')
    85→  if (parts.length >= 2) return (parts[0][0] + parts[1][0]).toUpperCase()
    86→  return clean.slice(0, 2).toUpperCase()
    87→}
    88→
    89→function getAvatarGradient(id: string): string {
    90→  let hash = 0
    91→  for (let i = 0; i < id.length; i++) {
    92→    hash = id.charCodeAt(i) + ((hash << 5) - hash)
    93→  }
    94→  return avatarGradients[Math.abs(hash) % avatarGradients.length]
    95→}

    97→function relativeDate(dateStr: string): string {
    98→  const date = new Date(dateStr)
    99→  const now = new Date()
   100→  const diffMs = now.getTime() - date.getTime()
   101→  const diffSec = Math.floor(diffMs / 1000)
   102→  const diffMin = Math.floor(diffSec / 60)
   103→  const diffHr = Math.floor(diffMin / 60)
   104→  const diffDay = Math.floor(diffHr / 24)
   105→  const diffMo = Math.floor(diffDay / 30)
   106→  const diffYr = Math.floor(diffDay / 365)
   107→
   108→  if (diffYr > 0) return `${diffYr}y ago`
   109→  if (diffMo > 0) return `${diffMo}mo ago`
   110→  if (diffDay > 0) return `${diffDay}d ago`
   111→  if (diffHr > 0) return `${diffHr}h ago`
   112→  if (diffMin > 0) return `${diffMin}m ago`
   113→  return 'Just now'
   114→}
