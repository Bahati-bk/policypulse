    91→function getFromStorage<T>(key: string, fallback: T): T {
    92→  if (typeof window === 'undefined') return fallback
    93→  try {
    94→    const raw = localStorage.getItem(key)
    95→    return raw ? (JSON.parse(raw) as T) : fallback
    96→  } catch {
    97→    return fallback
    98→  }
    99→}
   100→
   101→function saveToStorage(key: string, value: unknown) {
   102→  try {
   103→    localStorage.setItem(key, JSON.stringify(value))
   104→  } catch {
   105→    // ignore storage errors
   106→  }
   107→}

   114→  return <Monitor className="h-4 w-4" />
   115→}

   755→// --- Time options generator ---
   756→
   757→function generateTimeOptions(): string[] {
   758→  const times: string[] = []
   759→  for (let h = 0; h < 24; h++) {
   760→    for (let m = 0; m < 60; m += 30) {
   761→      const hh = h.toString().padStart(2, '0')
   762→      const mm = m.toString().padStart(2, '0')
   763→      times.push(`${hh}:${mm}`)
   764→    }
   765→  }
   766→  return times
   767→}
