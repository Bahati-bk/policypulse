    66→    66→function getInitials(name: string | null | undefined): string {
    67→    67→  if (!name) return 'U'
    68→    68→  const parts = name.split(' ')
    69→    69→  if (parts.length >= 2) return (parts[0][0] + parts[1][0]).toUpperCase()
    70→    70→  return name.slice(0, 2).toUpperCase()
    71→    71→}

    73→    73→function getAvatarGradient(id: string): string {
    74→    74→  let hash = 0
    75→    75→  for (let i = 0; i < id.length; i++) hash = id.charCodeAt(i) + ((hash << 5) - hash)
    76→    76→  const gradients = [
    77→    77→    'from-emerald-500 to-teal-600',
    78→    78→    'from-teal-500 to-cyan-600',
    79→    79→    'from-amber-500 to-orange-600',
    80→    80→    'from-rose-500 to-pink-600',
    81→    81→    'from-emerald-400 to-cyan-500',
    82→    82→  ]
    83→    83→  return gradients[Math.abs(hash) % gradients.length]
    84→    84→}
