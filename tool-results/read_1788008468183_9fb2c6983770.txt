   123→  changeType: string

   125→  whoIsAffected: string
   126→  whatToDo: string

   139→// ── Component ────────────────────────────────────────────────────────
   140→
   141→export default function AlertsView() {
   142→  const user = useAppStore(s => s.user)
   143→  const queryClient = useQueryClient()
   144→  const [statusFilter, setStatusFilter] = useState('')
   145→  const [search, setSearch] = useState('')
   146→  const [selectedAlert, setSelectedAlert] = useState<Alert | null>(null)
   147→  const [rejectDialog, setRejectDialog] = useState<string | null>(null)
   148→  const [rejectReason, setRejectReason] = useState('')
   149→  const [createOpen, setCreateOpen] = useState(false)
   150→  const [form, setForm] = useState<FormData>(emptyForm)
   151→
   152→  // ── Queries ──────────────────────────────────────────────────────
   153→
   154→  const { data: alertsData, isLoading } = useQuery<Alert[]>({
   155→    queryKey: ['alerts', statusFilter],
   156→    queryFn: () => fetch(`/api/alerts${statusFilter ? `?status=${statusFilter}` : ''}`).then(r => r.json()),
   157→    enabled: !!user,
   158→  })
   159→  const alerts = safeArray<Alert>(alertsData)
   160→
   161→  // ── Counts for filter badges ─────────────────────────────────────
   162→
   163→  const statusCounts = useMemo(() => {
   164→    const counts: Record<string, number> = { '': alerts.length }
   165→    for (const a of alerts) {
   166→      counts[a.status] = (counts[a.status] || 0) + 1
   167→    }
   168→    return counts
   169→  }, [alerts])
   170→
   171→  // ── Mutations ────────────────────────────────────────────────────
   172→
   173→  const createMutation = useMutation({
   174→    mutationFn: async (data: FormData) => {
   175→      const res = await fetch('/api/alerts', {
   176→        method: 'POST',
   177→        headers: { 'Content-Type': 'application/json' },
   178→        body: JSON.stringify(data),
   179→      })
   180→      if (!res.ok) {
   181→        const d = await res.json()
   182→        throw new Error(d.error || 'Failed to create alert')
   183→      }
   184→      return res.json()
   185→    },
   186→    onSuccess: () => {
   187→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   188→      toast.success('Alert created successfully')
   189→      setCreateOpen(false)
   190→      setForm(emptyForm)
   191→    },
   192→    onError: (e) => toast.error(e.message),
   193→  })
   194→
   195→  const approveMutation = useMutation({
   196→    mutationFn: async (id: string) => {
   197→      const res = await fetch(`/api/alerts/${id}/approve`, { method: 'POST' })
   198→      if (!res.ok) throw new Error('Failed')
   199→      return res.json()
   200→    },
   201→    onSuccess: () => {
   202→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   203→      toast.success('Alert approved')
   204→      setSelectedAlert(null)
   205→    },
   206→    onError: () => toast.error('Failed to approve'),
   207→  })
   208→
   209→  const rejectMutation = useMutation({
   210→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   211→      const res = await fetch(`/api/alerts/${id}/reject`, {
   212→        method: 'POST',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({ reason }),
   215→      })
   216→      if (!res.ok) throw new Error('Failed')
   217→      return res.json()
   218→    },
   219→    onSuccess: () => {
   220→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   221→      toast.success('Alert rejected')
   222→      setRejectDialog(null)
   223→      setSelectedAlert(null)
   224→    },
   225→    onError: () => toast.error('Failed to reject'),
   226→  })
   227→
   228→  const sendMutation = useMutation({
   229→    mutationFn: async (id: string) => {
   230→      const res = await fetch(`/api/alerts/${id}/send`, { method: 'POST' })
   231→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Failed') }
   232→      return res.json()
   233→    },
   234→    onSuccess: (data) => {
   235→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   236→      queryClient.invalidateQueries({ queryKey: ['notifications'] })
   237→      toast.success(`Notifications sent to ${data.sentTo} users`)
   238→      setSelectedAlert(null)
   239→    },
   240→    onError: (e) => toast.error(e.message),
   241→  })
   242→
   243→  // ── Helpers ──────────────────────────────────────────────────────
   244→
   245→  const isAdmin = user?.role === 'ADMIN'
   246→
   247→  const filtered = alerts.filter((alert) => {
   248→    if (!search) return true
   249→    const q = search.toLowerCase()
   250→    return (
   251→      alert.title?.toLowerCase().includes(q) ||
   252→      alert.summary?.toLowerCase().includes(q) ||
   253→      alert.whatChanged?.toLowerCase().includes(q)
   254→    )
   255→  })
   256→
   257→  function updateField<K extends keyof FormData>(key: K, value: FormData[K]) {
   258→    setForm(prev => ({ ...prev, [key]: value }))
   259→  }
   260→
   261→  function resetForm() {
   262→    setForm(emptyForm)
   263→  }
   264→
   265→  function handleSubmit(e: React.FormEvent) {
   266→    e.preventDefault()
   267→    if (!form.title.trim() || form.summary.trim().length < 10) return
   268→    createMutation.mutate(form)
   269→  }
   270→
   271→  function exportCSV() {
   272→    const headers = ['Title', 'Status', 'Severity', 'Change Type', 'Created', 'Summary', 'What Changed', 'Who Is Affected']
   273→    const rows = filtered.map(a => [
   274→      `"${(a.title || '').replace(/"/g, '\"')}"`,
   275→      a.status,
   276→      a.severity || '',
   277→      a.changeType || '',
   278→      a.createdAt ? format(new Date(a.createdAt), 'yyyy-MM-dd') : '',
   279→      `"${(a.summary || '').replace(/"/g, '\"')}"`,
   280→      `"${(a.whatChanged || '').replace(/"/g, '\"')}"`,
   281→      `"${(a.whoIsAffected || '').replace(/"/g, '\"')}"`,
   282→    ])
   283→    const csv = [headers.join(','), ...rows.map(r => r.join(','))].join('\n')
   284→    const blob = new Blob([csv], { type: 'text/csv' })
   285→    const url = URL.createObjectURL(blob)
   286→    const a = document.createElement('a')
   287→    a.href = url
   288→    a.download = `policypulse-alerts-${format(new Date(), 'yyyy-MM-dd')}.csv`
   289→    a.click()
   290→    URL.revokeObjectURL(url)
   291→    toast.success('Alerts exported to CSV')
   292→  }
   293→
   294→  // ── Render ───────────────────────────────────────────────────────
   295→
   296→  return (
   297→    <motion.div variants={container} initial="hidden" animate="show" className="space-y-6">
   298→      {/* ── Header ── */}
   299→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   300→        <div>
   301→          <h1 className="text-2xl font-bold tracking-tight">Alerts</h1>
   302→          <p className="text-muted-foreground text-sm">Policy change alerts and notifications</p>
   303→        </div>
   304→        <div className="flex items-center gap-2">
   305→          {filtered.length > 0 && (
   306→            <Button variant="outline" size="sm" onClick={exportCSV}>
   307→              <Download className="mr-2 h-3.5 w-3.5" />Export CSV
   308→            </Button>
   309→          )}
   310→          <Button size="sm" onClick={() => setCreateOpen(true)}>
   311→            <Plus className="mr-2 h-3.5 w-3.5" />Create Alert
   312→          </Button>
   313→        </div>
   314→      </div>
   315→
   316→      {/* ── Filters ── */}
   317→      <div className="flex flex-col sm:flex-row gap-3">
   318→        <div className="relative flex-1">
   319→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   320→          <Input
   321→            placeholder="Search alerts by title, summary, or changes..."
   322→            value={search}
   323→            onChange={e => setSearch(e.target.value)}
   324→            className="pl-9"
   325→          />
   326→        </div>
   327→        <div className="flex gap-2 flex-wrap">
   328→          {statusFilterOptions.map(s => {
   329→            const count = statusCounts[s] || 0
   330→            return (
   331→              <Button
   332→                key={s}
   333→                variant={statusFilter === s ? 'default' : 'outline'}
   334→                size="sm"
   335→                onClick={() => setStatusFilter(s)}
   336→                className="gap-1.5"
   337→              >
   338→                {s ? s.replace(/_/g, ' ') : 'All'}
   339→                {count > 0 && (
   340→                  <span className={
   341→                    statusFilter === s
   342→                      ? 'bg-background/20 text-current text-xs px-1.5 py-0.5 rounded-full min-w-[20px] text-center'
   343→                      : 'bg-muted text-muted-foreground text-xs px-1.5 py-0.5 rounded-full min-w-[20px] text-center'
   344→                  }>
   345→                    {count}
   346→                  </span>
   347→                )}
   348→              </Button>
   349→            )
   350→          })}
   351→        </div>
   352→      </div>
   353→
   354→      {/* ── Alert List ── */}
   355→      {isLoading ? (
   356→        <div className="space-y-4">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-40 rounded-lg" />)}</div>
   357→      ) : filtered.length === 0 ? (
   358→        <Card>
   359→          <CardContent className="py-16 px-8 text-center">
   360→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-muted">
   361→              <Bell className="h-8 w-8 text-muted-foreground/50" />
   362→            </div>
   363→            <p className="font-semibold text-muted-foreground text-lg">No alerts found</p>
   364→            <p className="text-sm text-muted-foreground/70 mt-2 max-w-md mx-auto">
   365→              {search || statusFilter
   366→                ? 'Try adjusting your search or filter to find what you\'re looking for.'
   367→                : 'Alerts will appear here when policy changes are detected or manually created. Get started by creating your first alert.'}
   368→            </p>
   369→            {!search && !statusFilter && (
   370→              <Button className="mt-4" size="sm" onClick={() => setCreateOpen(true)}>
   371→                <Plus className="mr-2 h-3.5 w-3.5" />Create Alert
   372→              </Button>
   373→            )}
   374→          </CardContent>
   375→        </Card>
   376→      ) : (
   377→        <motion.div variants={container} className="space-y-3">
   378→          {filtered.map((alert) => (
   379→            <motion.div key={alert.id} variants={item}>
   380→              <Card
   381→                className={`hover:shadow-md transition-all cursor-pointer group border-l-4 ${severityBorder[alert.severity] || 'border-l-slate-300'}`}
   382→                onClick={() => setSelectedAlert(alert)}
   383→              >
   384→                <CardContent className="p-4">
   385→                  <div className="flex items-start justify-between gap-3">
   386→                    <div className="flex items-start gap-3 min-w-0">
   387→                      {/* Severity dot */}
   388→                      <div className="relative shrink-0 mt-1">
   389→                        <div className={`h-2.5 w-2.5 rounded-full ${severityColors[alert.severity] || 'bg-slate-400'}`} />
   390→                        {alert.severity === 'CRITICAL' && (
   391→                          <span className="absolute inset-0 rounded-full bg-rose-400 animate-ping opacity-75" />
   392→                        )}
   393→                      </div>
   394→                      <div className="min-w-0">
   395→                        <p className="font-medium text-sm">{alert.title}</p>
   396→                        {alert.summary && (
   397→                          <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{alert.summary}</p>
   398→                        )}
   399→                        <div className="flex items-center gap-2 mt-2 flex-wrap">
   400→                          <Badge variant="outline" className={statusColors[alert.status] || ''}>
   401→                            {alert.status.replace(/_/g, ' ')}
   402→                          </Badge>
   403→                          <Badge variant="outline" className={severityBadge[alert.severity] || ''}>
   404→                            {alert.severity}
   405→                          </Badge>
   406→                          {alert.changeType && (
   407→                            <Badge variant="outline" className={changeTypeBadge[alert.changeType] || ''}>
   408→                              {changeTypeLabel[alert.changeType] || alert.changeType.replace(/_/g, ' ')}
   409→                            </Badge>
   410→                          )}
   411→                        </div>
   412→                      </div>
   413→                    </div>
   414→                    <span className="text-xs text-muted-foreground shrink-0">
   415→                      {alert.createdAt ? format(new Date(alert.createdAt), 'MMM d, yyyy') : ''}
   416→                    </span>
   417→                  </div>
   418→                </CardContent>
   419→              </Card>
   420→            </motion.div>
   421→          ))}
   422→        </motion.div>
   423→      )}
   424→
   425→      {/* ── Create Alert Dialog ── */}
   426→      <Dialog open={createOpen} onOpenChange={(open) => { if (!open) { setCreateOpen(false); resetForm() } }}>
   427→        <DialogContent className="max-w-2xl max-h-[90vh] pointer-events-auto">
   428→          <DialogHeader>
   429→            <DialogTitle className="flex items-center gap-2">
   430→              <Zap className="h-5 w-5 text-amber-500" />
   431→              Create New Alert
   432→            </DialogTitle>
   433→            <DialogDescription>
   434→              Fill in the details to create a new policy change alert. The alert will be created as a draft.
   435→            </DialogDescription>
   436→          </DialogHeader>
   437→
   438→          <form onSubmit={handleSubmit}>
   439→            <ScrollArea className="max-h-[65vh] pr-4">
   440→              <div className="space-y-5 pb-4">
   441→                {/* Title */}
   442→                <div className="space-y-2">
   443→                  <Label htmlFor="alert-title">Title <span className="text-red-500">*</span></Label>
   444→                  <Input
   445→                    id="alert-title"
   446→                    placeholder="e.g. NEW: Digital Service Tax Amendment"
   447→                    value={form.title}
   448→                    onChange={e => updateField('title', e.target.value)}
   449→                    required
   450→                  />
   451→                </div>
   452→
   453→                {/* Summary */}
   454→                <div className="space-y-2">
   455→                  <div className="flex items-center justify-between">
   456→                    <Label htmlFor="alert-summary">Summary <span className="text-red-500">*</span></Label>
   457→                    <span className={`text-xs ${form.summary.length >= 10 ? 'text-muted-foreground' : 'text-amber-500'}`}>
   458→                      {form.summary.length}/10 min characters
   459→                    </span>
   460→                  </div>
   461→                  <Textarea
   462→                    id="alert-summary"
   463→                    placeholder="Brief summary of the policy change..."
   464→                    value={form.summary}
   465→                    onChange={e => updateField('summary', e.target.value)}
   466→                    rows={3}
   467→                    required
   468→                  />
   469→                </div>
   470→
   471→                {/* Severity + Change Type row */}
   472→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   473→                  <div className="space-y-2">
   474→                    <Label>Severity</Label>
   475→                    <Select value={form.severity} onValueChange={v => updateField('severity', v)}>
   476→                      <SelectTrigger className="w-full">
   477→                        <div className="flex items-center gap-2">
   478→                          <span className={`h-2.5 w-2.5 rounded-full shrink-0 ${severityColors[form.severity]}`} />
   479→                          <SelectValue />
   480→                        </div>
   481→                      </SelectTrigger>
   482→                      <SelectContent>
   483→                        {severityOptions.map(s => (
   484→                          <SelectItem key={s} value={s}>
   485→                            <span className="flex items-center gap-2">
   486→                              <span className={`h-2.5 w-2.5 rounded-full ${severityColors[s]}`} />
   487→                              {s.charAt(0) + s.slice(1).toLowerCase()}
   488→                            </span>
   489→                          </SelectItem>
   490→                        ))}
   491→                      </SelectContent>
   492→                    </Select>
   493→                  </div>
   494→
   495→                  <div className="space-y-2">
   496→                    <Label>Change Type</Label>
   497→                    <Select value={form.changeType} onValueChange={v => updateField('changeType', v)}>
   498→                      <SelectTrigger className="w-full">
   499→                        <SelectValue />
   500→                      </SelectTrigger>
   501→                      <SelectContent>
   502→                        {changeTypeOptions.map(ct => (
   503→                          <SelectItem key={ct} value={ct}>
   504→                            <span className="flex items-center gap-2">
   505→                              <span className={`h-2 w-2 rounded-sm ${changeTypeBadge[ct].split(' ')[0]}`} />
   506→                              {changeTypeLabel[ct]}

   631→                        </CardContent>
   632→                      </Card>

   700→                  <Card className="border-l-4 border-l-primary">
   701→                    <CardContent className="p-4">
   702→                      <div className="flex items-center gap-2 mb-2">

   720→                {isAdmin && (

   739→
   740→                {selectedAlert.rejectionReason && (
