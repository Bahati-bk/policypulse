   138→// ── Component ────────────────────────────────────────────────────────
   139→
   140→export default function AlertsView() {
   141→  const user = useAppStore(s => s.user)
   142→  const queryClient = useQueryClient()
   143→  const [statusFilter, setStatusFilter] = useState('')
   144→  const [search, setSearch] = useState('')
   145→  const [selectedAlert, setSelectedAlert] = useState<Alert | null>(null)
   146→  const [rejectDialog, setRejectDialog] = useState<string | null>(null)
   147→  const [rejectReason, setRejectReason] = useState('')
   148→  const [createOpen, setCreateOpen] = useState(false)
   149→  const [form, setForm] = useState<FormData>(emptyForm)
   150→
   151→  // ── Queries ──────────────────────────────────────────────────────
   152→
   153→  const { data: alerts = [], isLoading } = useQuery<Alert[]>({
   154→    queryKey: ['alerts', statusFilter],
   155→    queryFn: () => fetch(`/api/alerts${statusFilter ? `?status=${statusFilter}` : ''}`).then(r => r.json()),
   156→  })
   157→
   158→  // ── Counts for filter badges ─────────────────────────────────────
   159→
   160→  const statusCounts = useMemo(() => {
   161→    const counts: Record<string, number> = { '': alerts.length }
   162→    for (const a of alerts) {
   163→      counts[a.status] = (counts[a.status] || 0) + 1
   164→    }
   165→    return counts
   166→  }, [alerts])
   167→
   168→  // ── Mutations ────────────────────────────────────────────────────
   169→
   170→  const createMutation = useMutation({
   171→    mutationFn: async (data: FormData) => {
   172→      const res = await fetch('/api/alerts', {
   173→        method: 'POST',
   174→        headers: { 'Content-Type': 'application/json' },
   175→        body: JSON.stringify(data),
   176→      })
   177→      if (!res.ok) {
   178→        const d = await res.json()
   179→        throw new Error(d.error || 'Failed to create alert')
   180→      }
   181→      return res.json()
   182→    },
   183→    onSuccess: () => {
   184→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   185→      toast.success('Alert created successfully')
   186→      setCreateOpen(false)
   187→      setForm(emptyForm)
   188→    },
   189→    onError: (e) => toast.error(e.message),
   190→  })
   191→
   192→  const approveMutation = useMutation({
   193→    mutationFn: async (id: string) => {
   194→      const res = await fetch(`/api/alerts/${id}/approve`, { method: 'POST' })
   195→      if (!res.ok) throw new Error('Failed')
   196→      return res.json()
   197→    },
   198→    onSuccess: () => {
   199→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   200→      toast.success('Alert approved')
   201→      setSelectedAlert(null)
   202→    },
   203→    onError: () => toast.error('Failed to approve'),
   204→  })
   205→
   206→  const rejectMutation = useMutation({
   207→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   208→      const res = await fetch(`/api/alerts/${id}/reject`, {
   209→        method: 'POST',
   210→        headers: { 'Content-Type': 'application/json' },
   211→        body: JSON.stringify({ reason }),
   212→      })
   213→      if (!res.ok) throw new Error('Failed')
   214→      return res.json()
   215→    },
   216→    onSuccess: () => {
   217→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   218→      toast.success('Alert rejected')
   219→      setRejectDialog(null)
   220→      setSelectedAlert(null)
   221→    },
   222→    onError: () => toast.error('Failed to reject'),
   223→  })
   224→
   225→  const sendMutation = useMutation({
   226→    mutationFn: async (id: string) => {
   227→      const res = await fetch(`/api/alerts/${id}/send`, { method: 'POST' })
   228→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Failed') }
   229→      return res.json()
   230→    },
   231→    onSuccess: (data) => {
   232→      queryClient.invalidateQueries({ queryKey: ['alerts'] })
   233→      queryClient.invalidateQueries({ queryKey: ['notifications'] })
   234→      toast.success(`Notifications sent to ${data.sentTo} users`)
   235→      setSelectedAlert(null)
   236→    },
   237→    onError: (e) => toast.error(e.message),
   238→  })
   239→
   240→  // ── Helpers ──────────────────────────────────────────────────────
   241→
   242→  const isAdmin = user?.role === 'ADMIN'
   243→
   244→  const filtered = alerts.filter((alert) => {
   245→    if (!search) return true
   246→    const q = search.toLowerCase()
   247→    return (
   248→      alert.title?.toLowerCase().includes(q) ||
   249→      alert.summary?.toLowerCase().includes(q) ||
   250→      alert.whatChanged?.toLowerCase().includes(q)
   251→    )
   252→  })
   253→
   254→  function updateField<K extends keyof FormData>(key: K, value: FormData[K]) {
   255→    setForm(prev => ({ ...prev, [key]: value }))
   256→  }
   257→
   258→  function resetForm() {
   259→    setForm(emptyForm)
   260→  }
   261→
   262→  function handleSubmit(e: React.FormEvent) {
   263→    e.preventDefault()
   264→    if (!form.title.trim() || form.summary.trim().length < 10) return
   265→    createMutation.mutate(form)
   266→  }
   267→
   268→  function exportCSV() {
   269→    const headers = ['Title', 'Status', 'Severity', 'Change Type', 'Created', 'Summary', 'What Changed', 'Who Is Affected']
   270→    const rows = filtered.map(a => [
   271→      `"${(a.title || '').replace(/"/g, '\"')}"`,
   272→      a.status,
   273→      a.severity || '',
   274→      a.changeType || '',
   275→      a.createdAt ? format(new Date(a.createdAt), 'yyyy-MM-dd') : '',
   276→      `"${(a.summary || '').replace(/"/g, '\"')}"`,
   277→      `"${(a.whatChanged || '').replace(/"/g, '\"')}"`,
   278→      `"${(a.whoIsAffected || '').replace(/"/g, '\"')}"`,
   279→    ])
   280→    const csv = [headers.join(','), ...rows.map(r => r.join(','))].join('\n')
   281→    const blob = new Blob([csv], { type: 'text/csv' })
   282→    const url = URL.createObjectURL(blob)
   283→    const a = document.createElement('a')
   284→    a.href = url
   285→    a.download = `policypulse-alerts-${format(new Date(), 'yyyy-MM-dd')}.csv`
   286→    a.click()
   287→    URL.revokeObjectURL(url)
   288→    toast.success('Alerts exported to CSV')
   289→  }
   290→
   291→  // ── Render ───────────────────────────────────────────────────────
   292→
   293→  return (
   294→    <motion.div variants={container} initial="hidden" animate="show" className="space-y-6">
   295→      {/* ── Header ── */}
   296→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   297→        <div>
   298→          <h1 className="text-2xl font-bold tracking-tight">Alerts</h1>
   299→          <p className="text-muted-foreground text-sm">Policy change alerts and notifications</p>
   300→        </div>
   301→        <div className="flex items-center gap-2">
   302→          {filtered.length > 0 && (
   303→            <Button variant="outline" size="sm" onClick={exportCSV}>
   304→              <Download className="mr-2 h-3.5 w-3.5" />Export CSV
   305→            </Button>
   306→          )}
   307→          <Button size="sm" onClick={() => setCreateOpen(true)}>
   308→            <Plus className="mr-2 h-3.5 w-3.5" />Create Alert
   309→          </Button>
   310→        </div>
   311→      </div>
   312→
   313→      {/* ── Filters ── */}
   314→      <div className="flex flex-col sm:flex-row gap-3">
   315→        <div className="relative flex-1">
   316→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   317→          <Input
   318→            placeholder="Search alerts by title, summary, or changes..."
   319→            value={search}
   320→            onChange={e => setSearch(e.target.value)}
   321→            className="pl-9"
   322→          />
   323→        </div>
   324→        <div className="flex gap-2 flex-wrap">
   325→          {statusFilterOptions.map(s => {
   326→            const count = statusCounts[s] || 0
   327→            return (
   328→              <Button
   329→                key={s}
   330→                variant={statusFilter === s ? 'default' : 'outline'}
   331→                size="sm"
   332→                onClick={() => setStatusFilter(s)}
   333→                className="gap-1.5"
   334→              >
   335→                {s ? s.replace(/_/g, ' ') : 'All'}
   336→                {count > 0 && (
   337→                  <span className={
   338→                    statusFilter === s
   339→                      ? 'bg-background/20 text-current text-xs px-1.5 py-0.5 rounded-full min-w-[20px] text-center'
   340→                      : 'bg-muted text-muted-foreground text-xs px-1.5 py-0.5 rounded-full min-w-[20px] text-center'
   341→                  }>
   342→                    {count}
   343→                  </span>
   344→                )}
   345→              </Button>
   346→            )
   347→          })}
   348→        </div>
   349→      </div>
   350→
   351→      {/* ── Alert List ── */}
   352→      {isLoading ? (
   353→        <div className="space-y-4">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-40 rounded-lg" />)}</div>
   354→      ) : filtered.length === 0 ? (
   355→        <Card>
   356→          <CardContent className="py-16 px-8 text-center">
   357→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-muted">
   358→              <Bell className="h-8 w-8 text-muted-foreground/50" />
   359→            </div>
   360→            <p className="font-semibold text-muted-foreground text-lg">No alerts found</p>
   361→            <p className="text-sm text-muted-foreground/70 mt-2 max-w-md mx-auto">
   362→              {search || statusFilter
   363→                ? 'Try adjusting your search or filter to find what you\'re looking for.'
   364→                : 'Alerts will appear here when policy changes are detected or manually created. Get started by creating your first alert.'}
   365→            </p>
   366→            {!search && !statusFilter && (
   367→              <Button className="mt-4" size="sm" onClick={() => setCreateOpen(true)}>
   368→                <Plus className="mr-2 h-3.5 w-3.5" />Create Alert
   369→              </Button>
   370→            )}
   371→          </CardContent>
   372→        </Card>
   373→      ) : (
   374→        <motion.div variants={container} className="space-y-3">
   375→          {filtered.map((alert) => (
   376→            <motion.div key={alert.id} variants={item}>
   377→              <Card
   378→                className={`hover:shadow-md transition-all cursor-pointer group border-l-4 ${severityBorder[alert.severity] || 'border-l-slate-300'}`}
   379→                onClick={() => setSelectedAlert(alert)}
   380→              >
   381→                <CardContent className="p-4">
   382→                  <div className="flex items-start justify-between gap-3">
   383→                    <div className="flex items-start gap-3 min-w-0">
   384→                      {/* Severity dot */}
   385→                      <div className="relative shrink-0 mt-1">
   386→                        <div className={`h-2.5 w-2.5 rounded-full ${severityColors[alert.severity] || 'bg-slate-400'}`} />
   387→                        {alert.severity === 'CRITICAL' && (
   388→                          <span className="absolute inset-0 rounded-full bg-rose-400 animate-ping opacity-75" />
   389→                        )}
   390→                      </div>
   391→                      <div className="min-w-0">
   392→                        <p className="font-medium text-sm">{alert.title}</p>
   393→                        {alert.summary && (
   394→                          <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{alert.summary}</p>
   395→                        )}
   396→                        <div className="flex items-center gap-2 mt-2 flex-wrap">
   397→                          <Badge variant="outline" className={statusColors[alert.status] || ''}>
   398→                            {alert.status.replace(/_/g, ' ')}
   399→                          </Badge>
   400→                          <Badge variant="outline" className={severityBadge[alert.severity] || ''}>
   401→                            {alert.severity}
   402→                          </Badge>
   403→                          {alert.changeType && (
   404→                            <Badge variant="outline" className={changeTypeBadge[alert.changeType] || ''}>
   405→                              {changeTypeLabel[alert.changeType] || alert.changeType.replace(/_/g, ' ')}
   406→                            </Badge>
   407→                          )}
   408→                        </div>
   409→                      </div>
   410→                    </div>
   411→                    <span className="text-xs text-muted-foreground shrink-0">
   412→                      {alert.createdAt ? format(new Date(alert.createdAt), 'MMM d, yyyy') : ''}
   413→                    </span>
   414→                  </div>
   415→                </CardContent>
   416→              </Card>
   417→            </motion.div>
   418→          ))}
   419→        </motion.div>
   420→      )}
   421→
   422→      {/* ── Create Alert Dialog ── */}
   423→      <Dialog open={createOpen} onOpenChange={(open) => { if (!open) { setCreateOpen(false); resetForm() } }}>
   424→        <DialogContent className="max-w-2xl max-h-[90vh] pointer-events-auto">
   425→          <DialogHeader>
   426→            <DialogTitle className="flex items-center gap-2">
   427→              <Zap className="h-5 w-5 text-amber-500" />
   428→              Create New Alert
   429→            </DialogTitle>
   430→            <DialogDescription>
   431→              Fill in the details to create a new policy change alert. The alert will be created as a draft.
   432→            </DialogDescription>
   433→          </DialogHeader>
   434→
   435→          <form onSubmit={handleSubmit}>
   436→            <ScrollArea className="max-h-[65vh] pr-4">
   437→              <div className="space-y-5 pb-4">
   438→                {/* Title */}
   439→                <div className="space-y-2">
   440→                  <Label htmlFor="alert-title">Title <span className="text-red-500">*</span></Label>
   441→                  <Input
   442→                    id="alert-title"
   443→                    placeholder="e.g. NEW: Digital Service Tax Amendment"
   444→                    value={form.title}
   445→                    onChange={e => updateField('title', e.target.value)}
   446→                    required
   447→                  />
   448→                </div>
   449→
   450→                {/* Summary */}
   451→                <div className="space-y-2">
   452→                  <div className="flex items-center justify-between">
   453→                    <Label htmlFor="alert-summary">Summary <span className="text-red-500">*</span></Label>
   454→                    <span className={`text-xs ${form.summary.length >= 10 ? 'text-muted-foreground' : 'text-amber-500'}`}>
   455→                      {form.summary.length}/10 min characters
   456→                    </span>
   457→                  </div>
   458→                  <Textarea
   459→                    id="alert-summary"
   460→                    placeholder="Brief summary of the policy change..."
   461→                    value={form.summary}
   462→                    onChange={e => updateField('summary', e.target.value)}
   463→                    rows={3}
   464→                    required
   465→                  />
   466→                </div>
   467→
   468→                {/* Severity + Change Type row */}
   469→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   470→                  <div className="space-y-2">
   471→                    <Label>Severity</Label>
   472→                    <Select value={form.severity} onValueChange={v => updateField('severity', v)}>
   473→                      <SelectTrigger className="w-full">
   474→                        <div className="flex items-center gap-2">
   475→                          <span className={`h-2.5 w-2.5 rounded-full shrink-0 ${severityColors[form.severity]}`} />
   476→                          <SelectValue />
   477→                        </div>
   478→                      </SelectTrigger>
   479→                      <SelectContent>
   480→                        {severityOptions.map(s => (
   481→                          <SelectItem key={s} value={s}>
   482→                            <span className="flex items-center gap-2">
   483→                              <span className={`h-2.5 w-2.5 rounded-full ${severityColors[s]}`} />
   484→                              {s.charAt(0) + s.slice(1).toLowerCase()}
   485→                            </span>
   486→                          </SelectItem>
   487→                        ))}
   488→                      </SelectContent>
   489→                    </Select>
   490→                  </div>
   491→
   492→                  <div className="space-y-2">
   493→                    <Label>Change Type</Label>
   494→                    <Select value={form.changeType} onValueChange={v => updateField('changeType', v)}>
   495→                      <SelectTrigger className="w-full">
   496→                        <SelectValue />
   497→                      </SelectTrigger>
   498→                      <SelectContent>
   499→                        {changeTypeOptions.map(ct => (
   500→                          <SelectItem key={ct} value={ct}>
   501→                            <span className="flex items-center gap-2">
   502→                              <span className={`h-2 w-2 rounded-sm ${changeTypeBadge[ct].split(' ')[0]}`} />
   503→                              {changeTypeLabel[ct]}
   504→                            </span>
   505→                          </SelectItem>
   506→                        ))}
   507→                      </SelectContent>
   508→                    </Select>
