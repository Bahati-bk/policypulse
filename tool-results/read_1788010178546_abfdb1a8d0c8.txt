    44→function readFileAsBase64(file: File): Promise<string> {
    45→  return new Promise((resolve, reject) => {
    46→    const reader = new FileReader()
    47→    reader.onload = () => {
    48→      const result = reader.result as string
    49→      resolve(result.split(',')[1])
    50→    }
    51→    reader.onerror = reject
    52→    reader.readAsDataURL(file)
    53→  })
    54→}

   415→function ComparisonDetail({ comparison, open, onOpenChange }: { comparison: Record<string, unknown>; open: boolean; onOpenChange: (v: boolean) => void }) {
   416→  const [allExpanded, setAllExpanded] = useState(false)
   417→
   418→  const changes = useMemo(() => {
   419→    const raw = comparison.changes as Record<string, unknown>[] | undefined
   420→    return safeArray(raw)
   421→  }, [comparison])
   422→
   423→  const status = (comparison.status as string) || 'PENDING'
   424→  const cfg = statusConfig[status] || statusConfig.PENDING
   425→  const oldDoc = comparison.oldDocument as Record<string, unknown> | null
   426→  const newDoc = comparison.newDocument as Record<string, unknown> | null
   427→
   428→  const highCount = changes.filter(c => c.severity === 'HIGH').length
   429→  const medCount = changes.filter(c => c.severity === 'MEDIUM').length
   430→  const lowCount = changes.filter(c => c.severity === 'LOW').length
   431→
   432→  const grouped = useMemo(() => {
   433→    const g: Record<string, Record<string, unknown>[]> = {}
   434→    for (const c of changes) {
   435→      const t = (c.changeType as string) || 'MODIFIED'
   436→      if (!g[t]) g[t] = []
   437→      g[t].push(c)
   438→    }
   439→    return g
   440→  }, [changes])
   441→
   442→  return (
   443→    <Dialog open={open} onOpenChange={onOpenChange}>
   444→      <DialogContent className="max-w-3xl max-h-[85vh] overflow-hidden flex flex-col">
   445→        <DialogHeader>
   446→          <DialogTitle className="flex items-center gap-2">
