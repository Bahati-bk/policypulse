   153→   153→  const { data: alerts = [], isLoading } = useQuery<Alert[]>({

   172→   172→      const res = await fetch('/api/alerts', {
   173→   173→        method: 'POST',
   174→   174→        headers: { 'Content-Type': 'application/json' },
   175→   175→        body: JSON.stringify(data),
   176→   176→      })

   178→   178→        const d = await res.json()

   181→   181→      return res.json()

   193→   193→    mutationFn: async (id: string) => {

   196→   196→      return res.json()
   197→   197→    },

   209→   209→        method: 'POST',
