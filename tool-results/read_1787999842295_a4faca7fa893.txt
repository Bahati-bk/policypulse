     2→

     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { motion } from 'framer-motion'
     6→import { toast } from 'sonner'
     7→import { Card, CardContent } from '@/components/ui/card'
     8→import { Badge } from '@/components/ui/badge'
     9→import { Button } from '@/components/ui/button'
    10→import { Skeleton } from '@/components/ui/skeleton'
    11→import { Input } from '@/components/ui/input'
    12→import { Label } from '@/components/ui/label'
    13→import { Textarea } from '@/components/ui/textarea'
    14→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogTrigger } from '@/components/ui/dialog'
    15→import { ScrollArea } from '@/components/ui/scroll-area'
    16→import {
    17→  Select,
    18→  SelectContent,
    19→  SelectItem,
    20→  SelectTrigger,
    21→  SelectValue,
    22→} from '@/components/ui/select'
    23→import { FileText, Upload, Play, FolderOpen, Search, File, FileSpreadsheet, Loader2, LayoutGrid, List, X } from 'lucide-react'
    24→import { format } from 'date-fns'
    25→import { useAppStore } from '@/lib/store'
    26→import { safeArray } from '@/lib/safe-array'
    27→
    28→const statusColors: Record<string, string> = {
    29→  UPLOADED: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300',
    30→  VALIDATING: 'bg-amber-100 text-amber-700 dark:bg-amber-900 dark:text-amber-300',
    31→  PROCESSING: 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-400',
    32→  READY: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-400',
    33→  FAILED: 'bg-red-100 text-red-700 dark:bg-red-900/50 dark:text-red-400',
    34→}
    35→
    36→const statusIcons: Record<string, React.ReactNode> = {
    37→  UPLOADED: <FileText className="h-4 w-4 text-slate-400" />,
    38→  VALIDATING: <FileText className="h-4 w-4 text-amber-400" />,
    39→  PROCESSING: <Play className="h-4 w-4 text-amber-500 animate-pulse" />,
    40→  READY: <FileText className="h-4 w-4 text-emerald-500" />,
    41→  FAILED: <FileText className="h-4 w-4 text-red-500" />,
    42→}
    43→
    44→const ACCEPTED_EXTENSIONS = ['.pdf', '.docx', '.doc', '.txt']
    45→const ACCEPTED_TYPES = ['application/pdf', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', 'application/msword', 'text/plain']
    46→
    47→function getFileIcon(fileName: string) {
    48→  const ext = fileName?.split('.').pop()?.toLowerCase()
    49→  if (ext === 'pdf') return <File className="h-4 w-4 text-rose-500" />
    50→  if (ext === 'docx' || ext === 'doc') return <FileSpreadsheet className="h-4 w-4 text-teal-500" />
    51→  if (ext === 'txt') return <FileText className="h-4 w-4 text-slate-500" />
    52→  return <FileText className="h-4 w-4 text-muted-foreground" />
    53→}
    54→
    55→function formatFileSize(bytes: number): string {
    56→  if (bytes < 1024) return bytes + ' B'
    57→  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB'
    58→  return (bytes / (1024 * 1024)).toFixed(1) + ' MB'
    59→}
    60→
    61→const container = { hidden: { opacity: 0 }, show: { opacity: 1, transition: { staggerChildren: 0.05 } } }
    62→const item = { hidden: { opacity: 0, y: 10 }, show: { opacity: 1, y: 0 } }
    63→
    64→interface PolicyOption {
    65→  id: string
    66→  title: string
    67→}
    68→
    69→export default function DocumentsView() {
    70→  const queryClient = useQueryClient()
    71→  const user = useAppStore((s) => s.user)
    72→  const [selectedDoc, setSelectedDoc] = useState<Record<string, unknown> | null>(null)
    73→  const [uploadOpen, setUploadOpen] = useState(false)
    74→  const [search, setSearch] = useState('')
    75→  const [statusFilter, setStatusFilter] = useState('')
    76→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid')
    77→
    78→  // Upload form state
    79→  const [selectedFile, setSelectedFile] = useState<File | null>(null)
    80→  const [isDragOver, setIsDragOver] = useState(false)
    81→  const [formTitle, setFormTitle] = useState('')
    82→  const [formPolicyId, setFormPolicyId] = useState('')
    83→  const [formVersion, setFormVersion] = useState('')
    84→  const [formDescription, setFormDescription] = useState('')
    85→  const [formDate, setFormDate] = useState('')
    86→  const fileInputRef = useRef<HTMLInputElement>(null)
    87→
    88→  const resetUploadForm = useCallback(() => {
    89→    setSelectedFile(null)
    90→    setFormTitle('')
    91→    setFormPolicyId('')
    92→    setFormVersion('')
    93→    setFormDescription('')
    94→    setFormDate('')
    95→  }, [])
    96→
    97→  const { data: docsData, isLoading } = useQuery({
    98→    queryKey: ['documents'],
    99→    queryFn: () => fetch('/api/documents').then(r => r.json()),
   100→    enabled: !!user,
   101→  })
   102→  const docs = safeArray(docsData)
   103→
   104→  const { data: policiesData } = useQuery<PolicyOption[]>({
   105→    queryKey: ['policies-list'],
   106→    queryFn: () => fetch('/api/policies').then(r => r.json()),
   107→    enabled: !!user,
   108→  })
   109→  const policies = safeArray<PolicyOption>(policiesData)
   110→
   111→  const uploadMutation = useMutation({
   112→    mutationFn: async (formData: FormData) => {
   113→      const res = await fetch('/api/documents', { method: 'POST', body: formData })
   114→      if (!res.ok) {
   115→        const err = await res.json()
   116→        throw new Error(err.error || 'Upload failed')
   117→      }
   118→      return res.json()
   119→    },
   120→    onSuccess: () => {
   121→      queryClient.invalidateQueries({ queryKey: ['documents'] })
   122→      toast.success('Document uploaded successfully')
   123→      setUploadOpen(false)
   124→      resetUploadForm()
   125→    },
   126→    onError: (err: Error) => toast.error(err.message),
   127→  })
   128→
   129→  const processMutation = useMutation({
   130→    mutationFn: async (id: string) => {
   131→      const res = await fetch(`/api/documents/${id}/process`, { method: 'POST' })
   132→      if (!res.ok) throw new Error('Processing failed')
   133→      return res.json()
   134→    },
   135→    onSuccess: () => {
   136→      queryClient.invalidateQueries({ queryKey: ['documents'] })
   137→      toast.success('Document processed successfully')
   138→    },
   139→    onError: () => toast.error('Processing failed'),
   140→  })
   141→
   142→  const validateFile = useCallback((file: File): boolean => {
   143→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   144→    if (!ACCEPTED_EXTENSIONS.includes(ext)) {
   145→      toast.error('Unsupported file type. Please upload PDF, DOCX, DOC, or TXT files.')
   146→      return false
   147→    }
   148→    return true
   149→  }, [])
   150→
   151→  const handleFileSelect = useCallback((file: File) => {
   152→    if (!validateFile(file)) return
   153→    setSelectedFile(file)
   154→    // Auto-fill title from filename if empty
   155→    if (!formTitle) {
   156→      const nameWithoutExt = file.name.replace(/\.[^.]+$/, '')
   157→      setFormTitle(nameWithoutExt)
   158→    }
   159→  }, [validateFile, formTitle])
   160→
   161→  const handleDragOver = useCallback((e: React.DragEvent) => {
   162→    e.preventDefault()
   163→    e.stopPropagation()
   164→    setIsDragOver(true)
   165→  }, [])
   166→
   167→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   168→    e.preventDefault()
   169→    e.stopPropagation()
   170→    setIsDragOver(false)
   171→  }, [])
   172→
   173→  const handleDrop = useCallback((e: React.DragEvent) => {
   174→    e.preventDefault()
   175→    e.stopPropagation()
   176→    setIsDragOver(false)
   177→    const file = e.dataTransfer.files[0]
   178→    if (file) handleFileSelect(file)
   179→  }, [handleFileSelect])
   180→
   181→  const handleFileInput = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
   182→    const file = e.target.files?.[0]
   183→    if (file) handleFileSelect(file)
   184→  }, [handleFileSelect])
   185→
   186→  const handleUpload = (e: React.FormEvent<HTMLFormElement>) => {
   187→    e.preventDefault()
   188→    if (!selectedFile) {
   189→      toast.error('Please select a file to upload')
   190→      return
   191→    }
   192→    if (!formPolicyId) {
   193→      toast.error('Please select a policy')
   194→      return
   195→    }
   196→    const formData = new FormData()
   197→    formData.append('file', selectedFile)
   198→    formData.append('title', formTitle)
   199→    formData.append('policyId', formPolicyId)
   200→    if (formVersion) formData.append('version', formVersion)
   201→    if (formDescription) formData.append('description', formDescription)
   202→    if (formDate) formData.append('documentDate', formDate)
   203→    uploadMutation.mutate(formData)
   204→  }
   205→
   206→  const filteredDocs = docs
   207→    .filter((doc: Record<string, unknown>) => {
   208→      if (!search) return true
   209→      const q = search.toLowerCase()
   210→      return (
   211→        ((doc.policy as Record<string, unknown>)?.title as string)?.toLowerCase().includes(q) ||
   212→        (doc.version as string)?.toLowerCase().includes(q) ||
   213→        (doc.fileName as string)?.toLowerCase().includes(q)
   214→      )
   215→    })
   216→    .filter((doc: Record<string, unknown>) => !statusFilter || (doc.processingStatus as string) === statusFilter)
   217→
   218→  // Status counts - computed from docs
   219→  const statusCounts = (() => {
   220→    const counts: Record<string, number> = { '': docs.length }
   221→    for (const doc of docs as Record<string, unknown>[]) {
   222→      const s = (doc.processingStatus as string) || 'UPLOADED'
   223→      counts[s] = (counts[s] || 0) + 1
   224→    }
   225→    return counts
   226→  })()
   227→
   228→  return (
   229→    <motion.div variants={container} initial="hidden" animate="show" className="space-y-6">
   230→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   231→        <div>
   232→          <h1 className="text-2xl font-bold tracking-tight">Documents</h1>
   233→          <p className="text-muted-foreground text-sm">Manage policy documents and versions</p>
   234→        </div>
   235→        <div className="flex items-center gap-2">
   236→          <Dialog open={uploadOpen} onOpenChange={(open) => {
   237→            setUploadOpen(open)
   238→            if (!open) resetUploadForm()
   239→          }}>
   240→            <DialogTrigger asChild>
   241→              <Button>
   242→                <Upload className="mr-2 h-4 w-4" />
   243→                Upload Document
   244→                <Badge variant="secondary" className="ml-2 h-5 min-w-[20px] px-1.5 text-xs">
   245→                  {docs.length}
   246→                </Badge>
   247→              </Button>
   248→            </DialogTrigger>
   249→            <DialogContent className="max-w-lg max-h-[90vh]">
   250→              <DialogHeader>
   251→                <DialogTitle className="flex items-center gap-2">
   252→                  <Upload className="h-5 w-5 text-primary" />
   253→                  Upload Document
   254→                </DialogTitle>
   255→              </DialogHeader>
   256→              <ScrollArea className="max-h-[80vh] pr-4">
   257→                <form onSubmit={handleUpload} className="space-y-4">
   258→                  {/* Drag and Drop Zone */}
   259→                  <div className="space-y-2">
   260→                    <Label>File <span className="text-destructive">*</span></Label>
   261→                    <div
   262→                      role="button"
   263→                      tabIndex={0}
   264→                      onClick={() => fileInputRef.current?.click()}
   265→                      onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') fileInputRef.current?.click() }}
   266→                      onDragOver={handleDragOver}
   267→                      onDragLeave={handleDragLeave}
   268→                      onDrop={handleDrop}
   269→                      className={`
   270→                        relative flex flex-col items-center justify-center min-h-[200px] rounded-lg border-2 border-dashed p-6
   271→                        transition-colors cursor-pointer
   272→                        ${isDragOver
   273→                          ? 'border-primary bg-primary/10'
   274→                          : selectedFile
   275→                            ? 'border-emerald-500/50 bg-emerald-50 dark:bg-emerald-950/20'
   276→                            : 'border-muted-foreground/25 hover:border-muted-foreground/50 hover:bg-muted/50'
   277→                        }
   278→                      `
   279→                      }
   280→                    >
   281→                      <input
   282→                        ref={fileInputRef}
   283→                        type="file"
   284→                        accept=".pdf,.docx,.doc,.txt"
   285→                        className="hidden"
   286→                        onChange={handleFileInput}
   287→                      />
   288→
   289→                      {selectedFile ? (
   290→                        <div className="flex items-center gap-3 text-sm">
   291→                          <div className="h-10 w-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center shrink-0">
   292→                            {getFileIcon(selectedFile.name)}
   293→                          </div>
   294→                          <div className="text-left min-w-0">
   295→                            <p className="font-medium truncate max-w-[280px]">{selectedFile.name}</p>
   296→                            <p className="text-xs text-muted-foreground">{formatFileSize(selectedFile.size)}</p>
   297→                          </div>
   298→                          <Button
   299→                            type="button"
   300→                            variant="ghost"
   301→                            size="icon"
   302→                            className="h-8 w-8 shrink-0"
   303→                            onClick={(e) => {
   304→                              e.stopPropagation()
   305→                              setSelectedFile(null)
   306→                              if (fileInputRef.current) fileInputRef.current.value = ''
   307→                            }}
   308→                          >
   309→                            <X className="h-4 w-4" />
   310→                          </Button>
   311→                        </div>
   312→                      ) : (
   313→                        <>
   314→                          <div className={`h-12 w-12 rounded-full flex items-center justify-center mb-3 ${isDragOver ? 'bg-primary/20' : 'bg-muted'}`}>
   315→                            <Upload className={`h-6 w-6 ${isDragOver ? 'text-primary' : 'text-muted-foreground'}`} />
   316→                          </div>
   317→                          <p className="text-sm font-medium text-muted-foreground">
   318→                            {isDragOver ? 'Drop your file here' : 'Drag & drop a file here'}
   319→                          </p>
   320→                          <p className="text-xs text-muted-foreground/70 mt-1">
   321→                            Supports PDF, DOCX, DOC, TXT
   322→                          </p>
   323→                        </>
   324→                      )}
   325→                    </div>
   326→                    <p className="text-xs text-muted-foreground text-center">
   327→                      <button
   328→                        type="button"
   329→                        className="text-primary hover:underline"
   330→                        onClick={() => fileInputRef.current?.click()}
   331→                      >
   332→                        or browse files
   333→                      </button>
   334→                    </p>
   335→                  </div>
   336→
   337→                  {/* Title */}
   338→                  <div className="space-y-2">
   339→                    <Label htmlFor="doc-title">Title <span className="text-destructive">*</span></Label>
   340→                    <Input
   341→                      id="doc-title"
   342→                      value={formTitle}
   343→                      onChange={e => setFormTitle(e.target.value)}
   344→                      placeholder="Document title"
   345→                      required
   346→                    />
   347→                  </div>
   348→
   349→                  {/* Policy */}
   350→                  <div className="space-y-2">
   351→                    <Label>Policy <span className="text-destructive">*</span></Label>
   352→                    <Select value={formPolicyId} onValueChange={setFormPolicyId}>
   353→                      <SelectTrigger>
   354→                        <SelectValue placeholder="Select a policy" />
   355→                      </SelectTrigger>
   356→                      <SelectContent>
   357→                        {policies.map(p => (
   358→                          <SelectItem key={p.id} value={p.id}>{p.title}</SelectItem>
   359→                        ))}
   360→                      </SelectContent>
   361→                    </Select>
   362→                  </div>
   363→
   364→                  {/* Version */}
   365→                  <div className="space-y-2">
   366→                    <Label htmlFor="doc-version">Version</Label>
   367→                    <Input
   368→                      id="doc-version"
   369→                      value={formVersion}
   370→                      onChange={e => setFormVersion(e.target.value)}
   371→                      placeholder="e.g. 2024-Amendment"
   372→                    />
   373→                  </div>
   374→
   375→                  {/* Description */}
   376→                  <div className="space-y-2">
   377→                    <Label htmlFor="doc-desc">Description</Label>
   378→                    <Textarea
   379→                      id="doc-desc"
   380→                      value={formDescription}
   381→                      onChange={e => setFormDescription(e.target.value)}
   382→                      placeholder="Optional description..."
   383→                      rows={2}
   384→                    />
   385→                  </div>
   386→
   387→                  {/* Document Date */}
   388→                  <div className="space-y-2">
   389→                    <Label htmlFor="doc-date">Document Date</Label>
   390→                    <Input
   391→                      id="doc-date"
   392→                      type="date"
   393→                      value={formDate}
   394→                      onChange={e => setFormDate(e.target.value)}
   395→                    />
   396→                  </div>
   397→
   398→                  {/* Submit */}
   399→                  <Button type="submit" className="w-full" disabled={uploadMutation.isPending || !selectedFile}>
   400→                    {uploadMutation.isPending && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   401→                    {uploadMutation.isPending ? 'Uploading...' : 'Upload Document'}
   402→                  </Button>
   403→                </form>
   404→              </ScrollArea>
   405→            </DialogContent>
   406→          </Dialog>
   407→        </div>
   408→      </div>
   409→
   410→      <div className="flex flex-col sm:flex-row gap-3">
   411→        <div className="relative flex-1">
   412→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   413→          <Input
   414→            placeholder="Search by title, version, or filename..."
   415→            value={search}
   416→            onChange={e => setSearch(e.target.value)}
   417→            className="pl-9"
   418→          />
   419→        </div>
   420→        <div className="flex items-center gap-2">
   421→          <div className="flex gap-1.5 flex-wrap">
   422→            {['', 'UPLOADED', 'PROCESSING', 'READY', 'FAILED'].map(s => (
   423→              <Button
   424→                key={s}
   425→                variant={statusFilter === s ? 'default' : 'outline'}
   426→                size="sm"
   427→                onClick={() => setStatusFilter(s)}
   428→                className="gap-1.5"
   429→              >
   430→                {s || 'All'}
   431→                <Badge variant={statusFilter === s ? 'secondary' : 'outline'} className="h-4 min-w-[16px] px-1 text-[10px]">
   432→                  {statusCounts[s] || 0}
   433→                </Badge>
   434→              </Button>
   435→            ))}
   436→          </div>
   437→          <div className="hidden sm:flex items-center border rounded-md">
   438→            <Button
   439→              variant={viewMode === 'grid' ? 'secondary' : 'ghost'}
   440→              size="icon"
   441→              className="h-8 w-8 rounded-r-none"
   442→              onClick={() => setViewMode('grid')}
   443→            >
   444→              <LayoutGrid className="h-4 w-4" />
   445→            </Button>
   446→            <Button
   447→              variant={viewMode === 'list' ? 'secondary' : 'ghost'}
   448→              size="icon"
   449→              className="h-8 w-8 rounded-l-none"
   450→              onClick={() => setViewMode('list')}
   451→            >
   452→              <List className="h-4 w-4" />
   453→            </Button>
   454→          </div>
   455→        </div>
   456→      </div>
   457→
   458→      {isLoading ? (
   459→        <div className="grid gap-4 md:grid-cols-2">
   460→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-40 rounded-lg" />)}
   461→        </div>
   462→      ) : filteredDocs.length === 0 ? (
   463→        <Card>
   464→          <CardContent className="p-12 text-center">
   465→            <FolderOpen className="h-12 w-12 mx-auto mb-3 text-muted-foreground/40" />
   466→            <p className="font-medium text-muted-foreground">No documents found</p>
   467→            <p className="text-sm text-muted-foreground/70 mt-1">
   468→              {search || statusFilter ? 'Try adjusting your search or filter criteria.' : 'Upload your first document to get started.'}
   469→            </p>
   470→          </CardContent>
   471→        </Card>
   472→      ) : viewMode === 'grid' ? (
   473→        <motion.div variants={container} className="grid gap-4 md:grid-cols-2">
   474→          {filteredDocs.map((doc: Record<string, unknown>) => (
   475→            <motion.div key={doc.id as string} variants={item}>
   476→              <Card className="hover:shadow-md transition-all cursor-pointer group" onClick={() => setSelectedDoc(doc)}>
   477→                <CardContent className="p-4">
   478→                  <div className="flex items-start justify-between gap-2">
   479→                    <div className="flex items-center gap-3 min-w-0">
   480→                      <div className="h-10 w-10 rounded-lg bg-primary/10 flex items-center justify-center shrink-0 group-hover:bg-primary/20 transition-colors">
   481→                        {(doc.processingStatus as string) === 'PROCESSING'
   482→                          ? statusIcons.PROCESSING
   483→                          : (doc.processingStatus as string) === 'FAILED'
   484→                            ? statusIcons.FAILED
   485→                            : getFileIcon(doc.fileName as string || '')}
   486→                      </div>
   487→                      <div className="min-w-0">
   488→                        <p className="font-medium text-sm truncate">{(doc.policy as Record<string, unknown>)?.title || 'Untitled'}</p>
   489→                        <p className="text-xs text-muted-foreground">Version: {doc.version as string || 'N/A'}</p>
   490→                      </div>
   491→                    </div>
   492→                    <Badge variant="outline" className={statusColors[(doc.processingStatus as string) || 'UPLOADED']}>
   493→                      {(doc.processingStatus as string) || 'UPLOADED'}
   494→                    </Badge>
   495→                  </div>
   496→                  <div className="flex items-center gap-3 mt-3 text-xs text-muted-foreground">
   497→                    <span className="truncate">{doc.fileName as string || 'No file'}</span>
   498→                    <span className="shrink-0">·</span>
   499→                    <span className="shrink-0">{doc.createdAt ? format(new Date(doc.createdAt as string), 'MMM d, yyyy') : ''}</span>
   500→                  </div>
   501→                </CardContent>
   502→              </Card>
   503→            </motion.div>
   504→          ))}
   505→        </motion.div>
   506→      ) : (
   507→        <motion.div variants={container} className="space-y-2">
   508→          {filteredDocs.map((doc: Record<string, unknown>) => (
   509→            <motion.div key={doc.id as string} variants={item}>
   510→              <Card
   511→                className="hover:shadow-md transition-all cursor-pointer group"
   512→                onClick={() => setSelectedDoc(doc)}
   513→              >
   514→                <CardContent className="p-3 flex items-center gap-4">
