    23→function getGreeting(): string {
    24→  const h = new Date().getHours()
    25→  if (h < 12) return 'Good Morning'
    26→  if (h < 18) return 'Good Afternoon'
    27→  return 'Good Evening'
    28→}

    30→function getSeverityColor(name: string) {
    31→  const colors: Record<string, string> = {
    32→    LOW: 'var(--chart-4)',
    33→    MEDIUM: 'var(--chart-3)',
    34→    HIGH: 'var(--chart-5)',
    35→  }
    36→  return colors[name] || '#888'
    37→}

    39→function getAlertStatusColor(status: string) {
    40→  const map: Record<string, string> = {
    41→    DRAFT: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300',
    42→    PENDING_REVIEW: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300',
    43→    APPROVED: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300',
    44→    REJECTED: 'bg-rose-100 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300',
    45→    SENT: 'bg-teal-100 text-teal-700 dark:bg-teal-900/40 dark:text-teal-300',
    46→  }
    47→  return map[status] || 'bg-muted text-muted-foreground'

    50→function getActivityColor(action: string) {
    51→  const map: Record<string, string> = {
    52→    CREATE: 'bg-emerald-500',
    53→    UPDATE: 'bg-amber-500',
    54→    APPROVE: 'bg-teal-500',
    55→    SEND: 'bg-teal-400',
    56→    DELETE: 'bg-rose-500',
    57→    REJECT: 'bg-rose-500',
    58→    ANALYZE: 'bg-amber-400',
    59→    LOGIN: 'bg-slate-400',
    60→    SEED: 'bg-primary',
    61→    PROCESS: 'bg-amber-400',
    62→    VIEW: 'bg-muted-foreground',
    63→  }
    64→  return map[action] || 'bg-muted-foreground'
    65→}

    67→function getActivityIcon(action: string) {
    68→  if (action === 'CREATE' || action === 'SEED') return '✨'
    69→  if (action === 'UPDATE' || action === 'APPROVE') return '🔄'
    70→  if (action === 'ANALYZE' || action === 'PROCESS') return '🧠'
    71→  if (action === 'LOGIN') return '🔑'
    72→  if (action === 'SEND') return '📤'
    73→  if (action === 'VIEW') return '👁'
    74→  if (action === 'DELETE' || action === 'REJECT') return '❌'
    75→  return '📌'
    76→}
