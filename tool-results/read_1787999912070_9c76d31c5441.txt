    68→function getInitials(name: string | null | undefined): string {
    69→  if (!name) return 'U'
    70→  const parts = name.split(' ')
    71→  if (parts.length >= 2) return (parts[0][0] + parts[1][0]).toUpperCase()
    72→  return name.slice(0, 2).toUpperCase()
    73→}

    75→function getAvatarGradient(id: string): string {
    76→  let hash = 0
    77→  for (let i = 0; i < id.length; i++) hash = id.charCodeAt(i) + ((hash << 5) - hash)
    78→  const gradients = [
    79→    'from-emerald-500 to-teal-600',
    80→    'from-teal-500 to-cyan-600',
    81→    'from-amber-500 to-orange-600',
    82→    'from-rose-500 to-pink-600',
    83→    'from-emerald-400 to-cyan-500',
    84→  ]
    85→  return gradients[Math.abs(hash) % gradients.length]
    86→}
